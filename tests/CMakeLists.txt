function(rewind_add_test name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE rewind::rewind)
    rewind_apply_flags(${name})
    add_test(NAME ${name} COMMAND ${name} ${ARGN})
endfunction()
