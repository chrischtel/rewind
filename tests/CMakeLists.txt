function(rewind_add_test name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE rewind::rewind)
    rewind_apply_flags(${name})
    add_test(NAME ${name} COMMAND ${name} ${ARGN})
endfunction()

rewind_add_test(core_test)
rewind_add_test(timeline_test)
rewind_add_test(assets_test)
rewind_add_test(world_test)
rewind_add_test(reflect_test)
rewind_add_test(ecs_test)

if(REWIND_BUILD_SAMPLES)
    rewind_add_test(hot_reload_test $<TARGET_FILE:bounce_hot>)
    add_dependencies(hot_reload_test bounce_hot)
endif()
